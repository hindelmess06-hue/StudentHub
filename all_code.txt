class AuditLogEntry {
  AuditLogEntry({
    required id,
    required this.userEmail,
    required this.action,
    required this.details,
    required this.timestamp,
  }) : id = id ?? 'log_${DateTime.now().millisecondsSinceEpoch}';

  final String id;
  final String userEmail;
  final String action;
  final String details;
  final DateTime timestamp;

  Map<String, dynamic> toJson() => {
    'id': id,
    'userEmail': userEmail,
    'action': action,
    'details': details,
    'timestamp': timestamp.toIso8601String(),
  };

  factory AuditLogEntry.fromJson(Map<String, dynamic> j) => AuditLogEntry(
    id: j['id'] as String?,
    userEmail: (j['userEmail'] as String?) ?? 'System',
    action: (j['action'] as String?) ?? 'Action',
    details: (j['details'] as String?) ?? '',
    timestamp: j['timestamp'] != null ? DateTime.parse(j['timestamp'] as String) : DateTime.now(),
  );
}

class Announcement {
  Announcement({
    required this.id,
    required this.title,
    required this.body,
    required this.author,
    required this.category,
    required this.date,
    this.pinned = false,
    this.custom = false,
    this.targetDepartment,
  });

  final String id;
  final String title;
  final String body;
  final String author;
  final AnnCategory category;
  final DateTime date;
  final bool pinned;
  final bool custom;
  final String? targetDepartment; // e.g. 'Computer Science', 'Engineering', 'Business'

  String get excerpt {
    final flat = body.replaceAll('\n', ' ').trim();
    return flat.length <= 110 ? flat : '${flat.substring(0, 107)}...';
  }

  Map<String, dynamic> toJson() => {
    'id': id,
    'title': title,
    'body': body,
    'author': author,
    'category': category.name,
    'date': date.toIso8601String(),
    'pinned': pinned,
    'targetDepartment': targetDepartment,
  };

  factory Announcement.fromJson(Map<String, dynamic> j) => Announcement(
    id: j['id'] as String,
    title: j['title'] as String,
    body: j['body'] as String,
    author: j['author'] as String,
    category: AnnCategory.values.firstWhere(
          (c) => c.name == j['category'],
      orElse: () => AnnCategory.campusLife,
    ),
    date: DateTime.parse(j['date'] as String),
    pinned: (j['pinned'] as bool?) ?? false,
    custom: true,
    targetDepartment: j['targetDepartment'] as String?,
  );
}

class CampusService {
  CampusService({
    required this.name,
    required this.description,
    required this.icon,
    required this.location,
    required this.openMinute,
    required this.closeMinute,
    required this.days,
    this.contact,
    this.overrideStatus,
    this.emergencyBanner,
  });

  final String name;
  final String description;
  final IconData icon;
  final String location;
  final int openMinute;
  final int closeMinute;
  final Set<int> days;
  final String? contact;
  final String? overrideStatus;
  final String? emergencyBanner;

  bool isOpenAt(DateTime t) {
    if (overrideStatus == 'closed' || overrideStatus == 'maintenance') return false;
    if (overrideStatus == 'open') return true;
    if (!days.contains(t.weekday)) return false;
    final m = t.hour * 60 + t.minute;
    return m >= openMinute && m < closeMinute;
  }

  String get hoursLabel {
    if (overrideStatus == 'closed') return 'Closed (Manual Override)';
    if (overrideStatus == 'maintenance') return 'Closed for Maintenance';
    String f(int m) =>
        '${(m ~/ 60).toString().padLeft(2, '0')}:${(m % 60).toString().padLeft(2, '0')}';
    final allDay = openMinute == 0 && closeMinute >= 1440;
    final range = allDay ? '24h' : '${f(openMinute)} -${f(closeMinute)}';
    final names = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    final sorted = days.toList()..sort();
    if (sorted.length == 7) return 'Every day, $range';
    return '${sorted.map((d) => names[d - 1]).join(', ')},$range';
  }
}

    if (RegExp(r'[A-Z]').hasMatch(s) && RegExp(r'[a-z]').hasMatch(s)) n++;
    if (RegExp(r'\d').hasMatch(s)) n++;
    if (RegExp(r'[^A-Za-z0-9]').hasMatch(s)) n++;
    return n;

class EventsScreen extends StatefulWidget {
  const EventsScreen({super.key});

  @override
  State<EventsScreen> createState() => _EventsScreenState();
}

class _EventsScreenState extends State<EventsScreen> {
  DateTime? _day; // null = all upcoming
  int _cat = 0; // 0 = All, 1..n = categories, last = Going

  static final List<String> _labels = [
    'All',
    ...EventCategory.values.map((c) => c.label),
    'Going',
  ];

  List<CampusEvent> _filtered(AppState app) {
    var list = app.upcomingEvents;
    if (_day != null) {
      list = list.where((e) => sameDay(e.start, _day!)).toList();
    }
    if (_cat == _labels.length - 1) {
      list = list.where((e) => app.goingEvents.contains(e.id)).toList();
    } else if (_cat > 0) {
      final c = EventCategory.values[_cat - 1];
      list = list.where((e) => e.category == c).toList();
    }
    return list;
  }

  @override
  Widget build(BuildContext context) {
    final app = context.watch<AppState>();
    final today = DateTime.now();
    final days = [
      for (var i = 0; i < 14; i++)
        DateTime(today.year, today.month, today.day + i),
    ];
    final items = _filtered(app);
    final upcoming = app.upcomingEvents;

    // Build list with day headers when browsing "All".
    final children = <Widget>[];
    DateTime? lastDay;
    var idx = 0;
    for (final e in items) {
      if (_day == null && (lastDay == null || !sameDay(lastDay, e.start))) {
        children.add(Padding(
          padding: EdgeInsets.only(top: lastDay == null ? 0 : 12, bottom: 10),
          child: Row(
            children: [
              Text(relDay(e.start),
                  style: AppText.display(16, color: AppColors.muted)),
              const SizedBox(width: 8),
              if (relDay(e.start) != fmtDay(e.start))
                Text(fmtDay(e.start),
                    style: AppText.body(12.5, color: AppColors.muted)),
            ],
          ),
        ));
        lastDay = e.start;
      }
      children.add(Padding(
        padding: const EdgeInsets.only(bottom: 12),
        child: FadeSlideIn(
          index: idx++,
          child: EventTile(event: e, showDay: false),
        ),
      ));
    }

    return Column(
      children: [
        ScreenHeader(
          title: 'Events',
          subtitle: '${upcoming.length} coming up',
        ),
        SizedBox(
          height: 84,
          child: ListView.separated(
            scrollDirection: Axis.horizontal,
            padding: const EdgeInsets.symmetric(horizontal: 20),
            itemCount: days.length + 1,
            separatorBuilder: (_, __) => const SizedBox(width: 10),
            itemBuilder: (_, i) {
              if (i == 0) {
                return _DayChip(
                  top: 'All',
                  bottom: '${upcoming.length}',
                  selected: _day == null,
                  hasEvents: false,
                  onTap: () => setState(() => _day = null),
                );
              }
              final d = days[i - 1];
              final has = upcoming.any((e) => sameDay(e.start, d));
              final sel = _day != null && sameDay(_day!, d);
              return _DayChip(
                top: i == 1 ? 'Today' : weekdayShort(d),
                bottom: '${d.day}',
                selected: sel,
                hasEvents: has,
                onTap: () => setState(() => _day = d),
              );
            },
          ),
        ),
        const SizedBox(height: 14),
        FilterChips(
          labels: _labels,
          selected: _cat,
          onSelected: (i) => setState(() => _cat = i),
        ),
        const SizedBox(height: 16),
        Expanded(
          child: items.isEmpty
              ? EmptyState(
                  icon: Icons.event_busy_rounded,
                  title: _cat == _labels.length - 1
                      ? "You're not going to anything yet"
                      : 'Nothing planned',
                  subtitle: _cat == _labels.length - 1
                      ? 'Open an event and tap "I\'m going" to build your agenda.'
                      : 'No events match this day and category.',
                )
              : ListView(
                  padding: const EdgeInsets.fromLTRB(20, 0, 20, 130),
                  children: children,
                ),
        ),
      ],
    );
  }
}

  String toString() => message;

/// One interface, two implementations:
///  * [FirebaseAuthService]  real accounts + real verification e-mails
///  * [DemoAuthService]      on-device simulation so the app runs instantly
abstract class AuthService {
  /// true  = user types a code (demo). false = user taps a link in an e-mail.
  bool get usesCodeEntry;

  /// Demo mode only: the code that a real e-mail would have contained.
  String? get demoCode;

  Future<AuthUser?> currentUser();
  Future<AuthUser> signUp({
    required String name,
    required String email,
    required String password,
  });
  Future<AuthUser> signIn({required String email, required String password});
  Future<void> sendVerification();
  Future<bool> checkVerified();
  Future<bool> submitCode(String code);
  Future<void> sendPasswordReset(String email);

  /// Removes an unverified account so the person can start over
  /// (for example after mistyping their e-mail).
  Future<void> discardUnverified();
  Future<void> signOut();
}

class FirebaseAuthService implements AuthService {
  fb.FirebaseAuth get _auth => fb.FirebaseAuth.instance;

  @override
  bool get usesCodeEntry => false;

  @override
  String? get demoCode => null;

  AuthUser? _map(fb.User? u) => u == null
      ? null
      : AuthUser(
    uid: u.uid,
    email: (u.email ?? '').toLowerCase(),
    name: u.displayName ?? '',
    verified: u.emailVerified,
    // TODO: We can later fetch extra claims or Firestore profile data for roles here
    role: 'student',
    assignedEntity: null,
  );

  @override
  Future<AuthUser?> currentUser() async {
    final u = _auth.currentUser;
    if (u == null) return null;
    try {
      await u.reload();
    } on fb.FirebaseAuthException catch (e) {
      // Account deleted or disabled elsewhere: drop the stale session.
      if (e.code == 'user-not-found' ||
          e.code == 'user-disabled' ||
          e.code == 'user-token-expired') {
        await _auth.signOut();
        return null;
      }
      // Offline etc.: keep the cached session.
    } catch (_) {}
    return _map(_auth.currentUser);
  }

  @override
  Future<AuthUser> signUp({
    required String name,
    required String email,
    required String password,
  }) async {
    try {
      final cred = await _auth.createUserWithEmailAndPassword(
        email: email,
        password: password,
      );
      final user = cred.user!;
      await user.updateDisplayName(name);
      await user.sendEmailVerification();
      await user.reload();
      return _map(_auth.currentUser)!;
    } on fb.FirebaseAuthException catch (e) {
      throw AuthException(_message(e));
    }
  }

  @override
  Future<AuthUser> signIn({
    required String email,
    required String password,
  }) async {
    try {
      final cred = await _auth.signInWithEmailAndPassword(
        email: email,
        password: password,
      );
      return _map(cred.user)!;
    } on fb.FirebaseAuthException catch (e) {
      throw AuthException(_message(e));
    }
  }

  @override
  Future<void> sendVerification() async {
    try {
      await _auth.currentUser?.sendEmailVerification();
    } on fb.FirebaseAuthException catch (e) {
      throw AuthException(_message(e));
    }
  }

  @override
  Future<bool> checkVerified() async {
    final u = _auth.currentUser;
    if (u == null) return false;
    await u.reload();
    return _auth.currentUser?.emailVerified ?? false;
  }

  @override
  Future<bool> submitCode(String code) async => false;

  @override
  Future<void> sendPasswordReset(String email) async {
    try {
      await _auth.sendPasswordResetEmail(email: email);
    } on fb.FirebaseAuthException catch (e) {
      // Do not reveal whether an account exists.
      if (e.code == 'user-not-found') return;
      throw AuthException(_message(e));
    }
  }

  @override
  Future<void> discardUnverified() async {
    final u = _auth.currentUser;
    try {
      if (u != null && !u.emailVerified) {
        await u.delete();
        return;
      }
    } catch (_) {
      // requires-recent-login etc. Fall through to a plain sign-out.
    }
    await _auth.signOut();
  }

  @override
  Future<void> signOut() => _auth.signOut();

  String _message(fb.FirebaseAuthException e) {
    switch (e.code) {
      case 'email-already-in-use':
        return 'An account with this email already exists. Try signing in.';
      case 'invalid-email':
        return "That doesn't look like a valid email address.";
      case 'weak-password':
        return 'Choose a stronger password (at least 8 characters).';
      case 'user-not-found':
      case 'wrong-password':
      case 'invalid-credential':
        return 'Incorrect email or password.';
      case 'user-disabled':
        return 'This account has been disabled. Contact IT Support.';
      case 'too-many-requests':
        return 'Too many attempts. Please wait a few minutes and try again.';
      case 'network-request-failed':
        return 'No connection. Check your internet and try again.';
      case 'operation-not-allowed':
        return 'Email sign-in is not enabled in Firebase yet (see README).';
      default:
        return 'Something went wrong (${e.code}). Please try again.';
    }
  }
}

/// Simulates accounts on the phone. NO real e-mail is sent and passwords are
/// only lightly hashed: this exists so you can try the whole flow instantly.
class DemoAuthService implements AuthService {
  DemoAuthService(this._prefs);

  final SharedPreferences _prefs;
  static const _kUsers = 'demo_users';
  static const _kSession = 'demo_session';

  String? _code;

  @override
  bool get usesCodeEntry => true;

  @override
  String? get demoCode => _code;

  Map<String, dynamic> _users() {
    final raw = _prefs.getString(_kUsers);
    if (raw == null) return {};
    return Map<String, dynamic>.from(jsonDecode(raw) as Map);
  }

  Future<void> _saveUsers(Map<String, dynamic> users) =>
      _prefs.setString(_kUsers, jsonEncode(users));

  String _hash(String s) {
    var h = 0x811c9dc5;
    for (final c in s.codeUnits) {
      h ^= c;
      h = (h * 0x01000193) & 0xFFFFFFFF;
    }
    return h.toRadixString(16);
  }

  AuthUser _toUser(String email, Map<String, dynamic> u) => AuthUser(
    uid: email,
    email: email,
    name: (u['name'] as String?) ?? '',
    verified: (u['verified'] as bool?) ?? false,
    role: (u['role'] as String?) ?? 'student',
    assignedEntity: u['assignedEntity'] as String?,
  );

  void _newCode() => _code = (Random.secure().nextInt(900000) + 100000).toString();

  @override
  Future<AuthUser?> currentUser() async {
    final email = _prefs.getString(_kSession);
    if (email == null) return null;
    final u = _users()[email];
    if (u == null) return null;
    return _toUser(email, Map<String, dynamic>.from(u as Map));
  }

  @override
  Future<AuthUser> signUp({
    required String name,
    required String email,
    required String password,
  }) async {
    await Future<void>.delayed(const Duration(milliseconds: 700));
    final users = _users();
    if (users.containsKey(email)) {
      throw AuthException(
          'An account with this email already exists. Try signing in.');
    }
    users[email] = {
      'name': name,
      'pass': _hash(password),
      'verified': false,
      'role': 'student',
      'assignedEntity': null,
    };
    await _saveUsers(users);
    await _prefs.setString(_kSession, email);
    _newCode();
    return _toUser(email, Map<String, dynamic>.from(users[email] as Map));
  }

  @override
  Future<AuthUser> signIn({
    required String email,
    required String password,
  }) async {
    await Future<void>.delayed(const Duration(milliseconds: 700));
    final raw = _users()[email];
    if (raw == null) throw AuthException('Incorrect email or password.');
    final u = Map<String, dynamic>.from(raw as Map);
    if (u['pass'] != _hash(password)) {
      throw AuthException('Incorrect email or password.');
    }
    await _prefs.setString(_kSession, email);
    final user = _toUser(email, u);
    if (!user.verified) _newCode();
    return user;
  }

  @override
  Future<void> sendVerification() async {
    await Future<void>.delayed(const Duration(milliseconds: 400));
    _newCode();
  }

  @override
  Future<bool> checkVerified() async {
    final u = await currentUser();
    return u?.verified ?? false;
  }

  @override
  Future<bool> submitCode(String code) async {
    await Future<void>.delayed(const Duration(milliseconds: 500));
    final email = _prefs.getString(_kSession);
    if (email == null || _code == null || code.trim() != _code) return false;
    final users = _users();
    final u = Map<String, dynamic>.from(users[email] as Map);
    u['verified'] = true;
    users[email] = u;
    await _saveUsers(users);
    _code = null;
    return true;
  }

  @override
  Future<void> sendPasswordReset(String email) async {
    await Future<void>.delayed(const Duration(milliseconds: 400));
  }

  @override
  Future<void> discardUnverified() async {
    final email = _prefs.getString(_kSession);
    if (email != null) {
      final users = _users();
      final u = users[email];
      if (u is Map && u['verified'] != true) {
        users.remove(email);
        await _saveUsers(users);
      }
    }
    await signOut();
  }

  @override
  Future<void> signOut() async {
    _code = null;
    await _prefs.remove(_kSession);
  }
}

import 'dart:math';

import '../app_config.dart';

/// Client-side rules for who may register.
///
/// IMPORTANT: this stops honest mistakes and gives instant feedback, but it
/// can be bypassed by anyone calling the backend directly. The real guarantee
/// comes from (1) the e-mail verification link and (2) backend rules that
/// check the verified e-mail domain. See README.md, "Hardening".
class EmailRules {
  static final RegExp _shape =
      RegExp(r'^[a-z0-9._%+\-]+@[a-z0-9\-]+(\.[a-z0-9\-]+)+$');

  static String normalize(String raw) => raw.trim().toLowerCase();

  static bool isAllowedDomain(String domain) => AppConfig.allowedDomains
      .any((d) => domain == d || domain.endsWith('.$d'));

  /// Returns null when the address is acceptable, otherwise a message that is
  /// safe to show to the user.
  static String? validate(String raw) {
    final email = normalize(raw);
    if (email.isEmpty) return 'Enter your university email.';

    final atCount = '@'.allMatches(email).length;
    if (atCount != 1 || !_shape.hasMatch(email) || email.contains('..')) {
      return "That doesn't look like a valid email address.";
    }
    final local = email.split('@').first;
    if (local.startsWith('.') || local.endsWith('.')) {
      return "That doesn't look like a valid email address.";
    }

    final domain = email.split('@').last;
    if (isAllowedDomain(domain)) return null;

    final suggestion = _closestAllowed(domain);
    if (suggestion != null) {
      return 'Did you mean @$suggestion? Only UEMF emails are accepted.';
    }
    return 'Only UEMF emails are accepted (name@${AppConfig.allowedDomains.first}).';
  }

  static String? _closestAllowed(String domain) {
    for (final d in AppConfig.allowedDomains) {
      if (_distance(domain, d) <= 3) return d;
    }
    return null;
  }

  /// Levenshtein edit distance, used to catch typos like "ueuromed.com".
  static int _distance(String a, String b) {
    final dp = List.generate(a.length + 1, (_) => List<int>.filled(b.length + 1, 0));
    for (var i = 0; i <= a.length; i++) {
      dp[i][0] = i;
    }
    for (var j = 0; j <= b.length; j++) {
      dp[0][j] = j;
    }
    for (var i = 1; i <= a.length; i++) {
      for (var j = 1; j <= b.length; j++) {
        final cost = a[i - 1] == b[j - 1] ? 0 : 1;
        dp[i][j] = min(
          min(dp[i - 1][j] + 1, dp[i][j - 1] + 1),
          dp[i - 1][j - 1] + cost,
        );
      }
    }
    return dp[a.length][b.length];
  }
}

/// Whole calendar days between [d] and today (negative = past).
/// Uses UTC dates so daylight-saving changes can't shift the result.
int dayDiff(DateTime d, [DateTime? from]) {
  final n = from ?? DateTime.now();
  return DateTime.utc(d.year, d.month, d.day)
      .difference(DateTime.utc(n.year, n.month, n.day))
      .inDays;
}

String relDay(DateTime d) {
  final n = dayDiff(d);
  if (n == 0) return 'Today';
  if (n == 1) return 'Tomorrow';
  if (n == -1) return 'Yesterday';
  if (n > 1 && n < 7) return _wdLong[d.weekday - 1];
  return fmtDay(d);
}

String timeAgo(DateTime d) {
  final diff = DateTime.now().difference(d);
  if (diff.inMinutes < 1) return 'just now';
  if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
  if (diff.inHours < 24) return '${diff.inHours}h ago';
  if (diff.inDays < 7) return '${diff.inDays}d ago';
  return fmtDay(d);
}

      pageBuilder: (_, __, ___) => page,

class _FadeSlideInState extends State<FadeSlideIn>
    with SingleTickerProviderStateMixin {
  late final AnimationController _c = AnimationController(
    vsync: this,
    duration: const Duration(milliseconds: 520),
  );
  late final Animation<double> _a =
      CurvedAnimation(parent: _c, curve: Curves.easeOutCubic);

  @override
  void initState() {
    super.initState();
    final delay = 55 * widget.index.clamp(0, 10);

  @override
  void dispose() {
    _c.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return FadeTransition(
      opacity: _a,
      child: SlideTransition(
        position: Tween<Offset>(
          begin: const Offset(0, 0.08),
          end: Offset.zero,
        ).animate(_a),
        child: widget.child,
      ),
    );
  }

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();

  SystemChrome.setEnabledSystemUIMode(SystemUiMode.edgeToEdge);
  SystemChrome.setSystemUIOverlayStyle(const SystemUiOverlayStyle(
    statusBarColor: Colors.transparent,
    statusBarIconBrightness: Brightness.light,
    systemNavigationBarColor: Colors.transparent,
    systemNavigationBarIconBrightness: Brightness.light,
  ));

  if (AppConfig.useFirebase) {
    await Firebase.initializeApp(options: DefaultFirebaseOptions.currentPlatform);
  }

  final prefs = await SharedPreferences.getInstance();
  final AuthService auth =
      AppConfig.useFirebase ? FirebaseAuthService() : DemoAuthService(prefs);

  final state = AppState(prefs, auth);
  await state.init();

  runApp(
    ChangeNotifierProvider<AppState>.value(
      value: state,
      child: const StudentHubApp(),
    ),
  );
}
